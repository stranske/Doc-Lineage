RAW=Path(__file__).parent
OUT=ROOT/'docs/evidence/issue-113-main-integration'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def counts(p):
 root=ET.parse(p).getroot();cases=list(root.iter('testcase'))
 return {'path':str(p.relative_to(RAW)),'tests':len(cases),'failures':sum(len(c.findall('failure')) for c in cases),'errors':sum(len(c.findall('error')) for c in cases),'skips':sum(len(c.findall('skipped')) for c in cases)}

for path in sorted(set(selected)):
 relative=path.relative_to(RAW);target=OUT/relative;target.parent.mkdir(parents=True,exist_ok=True)
 raw=path.read_bytes();public=raw
 if relative.as_posix() in ('mutation/disconnected.log','mutation/disconnected.xml'):
  public=re.sub(rb'[\t ]+(?=\r?$)',b'',raw,flags=re.M)
 target.write_bytes(public)
 map.append({'path':relative.as_posix(),'raw_sha256':sha(path),'public_sha256':sha(target),'transformation':'strip trailing horizontal whitespace only' if raw!=public else 'none'})
for name in ('run-proof.py','rerun-full.py','preflight.py','bundle.py','admission-control.py'):
 shutil.copyfile(RAW/name,OUT/(name+'.txt'))

binding['commands']=[f'{label}-command.json' for label in ('full','controls','exact','mutation','control','legal','navigation','lint','format','type-verifier','type-focused')]

coverage=ET.parse(RAW/'coverage.xml').getroot();percent=float(re.search(r'Total coverage: ([0-9.]+)%',(RAW/'full.log').read_text())[1])
native=[json.loads(l) for l in (RAW/'native.jsonl').read_text().splitlines()]
before=[r for r in native if r['phase']=='before'];after=[r for r in native if r['phase']=='after'];assert len(before)==len(after) and all(r['returncode']==0 for r in after)
summary={'revision':binding['revision'],'input_bindings':len(binding['inputs']),'full':full,'controls':counts(RAW/'controls.xml'),'exact':counts(RAW/'exact.xml'),'mutation_red':counts(RAW/'mutation/disconnected.xml'),'mutation_restored':counts(RAW/'mutation/restored.xml'),'coverage_percent':percent,'native_calls':len(after),'native_all_rc0':True}
