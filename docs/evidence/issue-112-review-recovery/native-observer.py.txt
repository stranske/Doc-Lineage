"""Observe real bundled helper invocations without changing argv or results."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time
from zipfile import ZipFile

_original_run = subprocess.run

def _run(*args, **kwargs):
    argv = args[0] if args else kwargs.get('args')
    if not isinstance(argv, (list, tuple)) or not argv or Path(str(argv[0])).name != 'redline':
        return _original_run(*args, **kwargs)
    path = Path(argv[0]).resolve()
    digest = hashlib.sha256(path.read_bytes()).hexdigest()
    if digest != '12b0955111e348543e55ac9fbc06599b3e79294781198c6fb1b46f3029867c36':
        raise RuntimeError(f'Unexpected native helper bytes: {path} {digest}')
    root = Path(os.environ['DOC129_NATIVE_EVIDENCE']) / f'{os.getpid()}-{time.time_ns()}'
    root.mkdir(parents=True)
    record = {'path': str(path), 'sha256': digest, 'argv': [str(a) for a in argv], 'pid': os.getpid(), 'before_ns': time.time_ns()}
    def snapshot(role, location):
        p = Path(location)
        if p.is_file():
            data = p.read_bytes()
            (root / f'{role}.docx').write_bytes(data)
            record[f'{role}_sha256'] = hashlib.sha256(data).hexdigest()
            try:
                with ZipFile(p) as z:
                    (root / f'{role}-document.xml').write_bytes(z.read('word/document.xml'))
            except Exception as e:
                record[f'{role}_xml_error'] = str(e)
    snapshot('original', argv[1]); snapshot('modified', argv[2])
    (root / 'before.json').write_text(json.dumps(record, indent=2))
    try:
        result = _original_run(*args, **kwargs)
        record['returncode'] = result.returncode
        for name in ['stdout','stderr']:
            value = getattr(result,name)
            (root / f'{name}.txt').write_bytes(value if isinstance(value,bytes) else (value or '').encode())
        return result
    except BaseException as e:
        record['error'] = repr(e)
        record['returncode'] = getattr(e,'returncode',None)
        raise
    finally:
        snapshot('output', argv[3])
        record['after_ns'] = time.time_ns()
        (root / 'result.json').write_text(json.dumps(record, indent=2))

subprocess.run = _run
